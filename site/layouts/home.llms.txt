{{- $summary := strings.TrimSpace (.RawContent | replaceRE `\s+` " ") -}}
{{- $lines := slice
  (printf "# %s" .Site.Title)
  ""
  (printf "> %s" $summary)
  ""
  (printf "Documentation version: %s. Source revision: %s." .Site.Params.version .Site.Params.sourceRef)
  "These pages come from the repository's docs/ sources. For version-specific behavior, use the"
  "documentation and code for your release. This index routes questions; follow its links for evidence."
  ""
  "## Interactive Bootstrap Protocol"
  ""
  "1. Choose an operating mode with the user: Declarative Taskpack Mode (simple single-file .loopwright/taskpack.yaml + GitHub Action/App for repository sync and PRs) or Managed Controller Mode (out-of-repo TrustedRegistry registry.yaml + SHA-256-pinned ControllerConfig controller.yaml + TaskContract for fine-grained CLI governance)."
  "2. Fetch the Markdown exports listed below for the chosen mode before generating YAML (Declarative Taskpack Walkthrough + Taskpack Manifest Reference + GitHub Action Configuration for Mode 1; Managed Controller Walkthrough + Task Contracts and Schemas + Controller and Trusted Registry for Mode 2)."
  "3. Inspect the target repository (including existing automatic synchronized workflows, release scripts, version manifests, and folder layout) and confirm upstream sources, writable/protected scope, verification checks, and PR publication settings interactively with the user."
  "4. Emit two-space Pretty YAML with apiVersion: loopwright.dev/v1alpha1, keep .loopwright/memory/<id> outside writable scope, and never embed raw secrets or untrusted ${ENV} interpolation inside taskpack.yaml."
-}}
{{- $index := resources.Get "doc-index/README.md" -}}
{{- $catalog := index (split (index (split $index.Content "## All pages\n") 1) "\n## ") 0 -}}
{{- range split $catalog "\n" -}}
  {{- if hasPrefix . "### " -}}
    {{- if ne . "### Contribute" -}}
      {{- $lines = $lines | append "" (printf "## %s" (strings.TrimPrefix "### " .)) "" -}}
    {{- end -}}
  {{- else if hasPrefix . "| [" -}}
    {{- $fields := split . "|" -}}
    {{- $link := strings.TrimSpace (index $fields 1) -}}
    {{- $target := replaceRE `^\[[^\]]*\]\(([^\)]+)\)$` "$1" $link -}}
    {{- $description := strings.TrimSpace (index $fields 2) -}}
    {{- if and (ne $target "development.md") (not (hasPrefix $target "contribute/")) -}}
      {{- with site.GetPage (printf "/docs/%s" $target) -}}
        {{- $entry := printf "- [%s](%s): %s" (partial "page-title.html" .) (.OutputFormats.Get "Markdown").RelPermalink $description -}}
        {{- $lines = $lines | append $entry -}}
      {{- else -}}{{ errorf "docs/README.md indexes a page not mounted in Hugo: %s" $target }}{{- end -}}
    {{- end -}}
  {{- end -}}
{{- end -}}
{{- $lines = $lines | append "" "## Optional" "" (printf "- [Repository navigation](https://github.com/gpustack/loopwright/blob/%s/docs/README.md): Module relationships, design records, code entry points and applicable skills for contributors." .Site.Params.sourceRef) -}}
{{- printf "%s\n" (delimit $lines "\n") -}}
