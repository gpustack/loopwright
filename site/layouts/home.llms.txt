{{- $summary := strings.TrimSpace (.RawContent | replaceRE `\s+` " ") -}}
{{- $lines := slice (printf "# %s" .Site.Title) "" (printf "> %s" $summary) "" (printf "Documentation version: %s. Source revision: %s." .Site.Params.version .Site.Params.sourceRef) "These pages come from the repository's docs/ sources. For version-specific behavior, use the" "documentation and code for your release. This index routes questions; follow its links for evidence." -}}
{{- $index := resources.Get "doc-index/README.md" -}}
{{- $catalog := index (split (index (split $index.Content "## All pages\n") 1) "\n## ") 0 -}}
{{- range split $catalog "\n" -}}
  {{- if hasPrefix . "### " -}}
    {{- $lines = $lines | append "" (printf "## %s" (strings.TrimPrefix "### " .)) "" -}}
  {{- else if hasPrefix . "| [" -}}
    {{- $fields := split . "|" -}}
    {{- $link := strings.TrimSpace (index $fields 1) -}}
    {{- $target := replaceRE `^\[[^\]]*\]\(([^\)]+)\)$` "$1" $link -}}
    {{- $description := strings.TrimSpace (index $fields 2) -}}
    {{- with site.GetPage (printf "/docs/%s" $target) -}}
      {{- $entry := printf "- [%s](%s): %s" (partial "page-title.html" .) (.OutputFormats.Get "Markdown").RelPermalink $description -}}
      {{- $lines = $lines | append $entry -}}
    {{- else -}}{{ errorf "docs/README.md indexes a page not mounted in Hugo: %s" $target }}{{- end -}}
  {{- end -}}
{{- end -}}
{{- $lines = $lines | append "" "## Optional" "" (printf "- [Repository navigation](https://github.com/thxcode/loopwright/blob/%s/docs/README.md): Module relationships, design records, code entry points and applicable skills for contributors." .Site.Params.sourceRef) -}}
{{- printf "%s\n" (delimit $lines "\n") -}}
