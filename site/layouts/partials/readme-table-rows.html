{{- $normalized := replace .content "\r\n" "\n" -}} {{- $headingPrefix := printf "## %s\n" .heading -}} {{- $section := "" -}} {{- if hasPrefix $normalized $headingPrefix -}} {{- $section = index (split (strings.TrimPrefix $headingPrefix $normalized) "\n## ") 0 -}} {{- else -}} {{- $parts := split $normalized (printf "\n%s" $headingPrefix) -}} {{- if lt (len $parts) 2 -}}{{ errorf "docs/README.md is missing '## %s'" .heading }}{{- end -}} {{- $section = index (split (index $parts 1) "\n## ") 0 -}} {{- end -}} {{- $items := slice -}} {{- range split $section "\n" -}} {{- $trimmed := strings.TrimSpace . -}} {{- if hasPrefix $trimmed "### " -}} {{- $items = $items | append (dict "kind" "heading" "title" (strings.TrimSpace (strings.TrimPrefix "### " $trimmed))) -}} {{- else if hasPrefix $trimmed "|" -}} {{- $fields := split $trimmed "|" -}} {{- if ge (len $fields) 4 -}} {{- $link := strings.TrimSpace (index $fields 1) -}} {{- if findRE `^\[[^\]]+\]\([^\)]+\)$` $link 1 -}} {{- $label := replaceRE `^\[([^\]]*)\]\([^\)]+\)$` "$1" $link -}} {{- $target := strings.TrimSpace (replaceRE `^\[[^\]]*\]\(([^\)]+)\)$` "$1" $link) -}} {{- $description := strings.TrimSpace (index $fields 2) -}} {{- $items = $items | append (dict "kind" "row" "label" $label "target" $target "description" $description) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $items -}}
